users table
	id -> int
	username  -> varchar
	password  -> varchar
	status -> enum
	role  -> enum
	created_at  -> timestamp

products table  
    id -> int
	name  -> varchar
	description  -> text
	quantity -> int
	price  -> decimal 10 digit 2 decimal places
	created_at  -> timestamp
    created_by -> FK int
    updated_at -> timestamp on update timestamp
